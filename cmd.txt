# Build both Bedrock Edition Gem Gallery Packs (Option 2 Imposter + Option 3 High-to-Low PolyMesh & Hybrid Splat):
uv run splat2mc build-gallery

# Headless 3DGS Spherical Imposter Renderer (no Blender required):
uv run splat2mc render-imposter Example1.ply -o bedrock/01_Imposter_Pack/Imposter_RP/textures/entity/spritesheet_0.png --icon-out bedrock/01_Imposter_Pack/Imposter_RP/textures/items/gem_0.png -c 16 -r 5 --res 160

# High-to-Low 3DGS-to-PolyMesh + Hybrid Tangent Splat Baker:
uv run splat2mc export-mesh Example2.ply --geo-out bedrock/02_Oriented_Splat_Pack/SplatRenderer_RP/models/entity/splat_cloud_1.geo.json --tex-out bedrock/02_Oriented_Splat_Pack/SplatRenderer_RP/textures/entity/splat_cloud_1.png -f 1500 --hybrid-splats 380

# Oriented Particle Stream Exporter (Pack 02 /scriptevent sample:run):
uv run python tools/export_oriented.py Example2.ply -n 5000
